def _sha256(path):
    """Hash the full bytes of a provenance input or output."""
    return hashlib.sha256(path.read_bytes()).hexdigest()


def _verify(root, hashes):
    """Reject any nonoriginal numerical input before importing source modules."""
    for name, expected in hashes.items():
        path = root / name
        if not path.is_file() or _sha256(path) != expected:
            raise RuntimeError(f"ERROR original source hash mismatch: {path}")


def _notebook_function(root, notebook, name):
    """Extract the complete original timestep function from its notebook cell."""
    content = json.loads((root / notebook).read_text())
    for cell in content["cells"]:
        code = "".join(cell["source"])
        if cell["cell_type"] == "code" and f"def {name}(" in code:
            return code
    raise RuntimeError(f"ERROR notebook function missing: {name}")


def _initial_state(root, name, dynamics=False):
    """Execute an original initializer, with documented dynamics adaptations."""
    code = (root / name).read_text()
    if dynamics:
        for before, after in (
            ("grid_len = 1024", "grid_len = 16"),
            ("use_circular_overlap = False", "use_circular_overlap = True"),
        ):
            if code.count(before) != 1:
                raise RuntimeError(f"ERROR original initializer marker missing: {before}")
            code = code.replace(before, after)
    scope = {"__name__": f"capture_{name[:-3]}", "__file__": str(root / name)}
    exec(compile(code, str(root / name), "exec"), scope)
    return scope["vs"], scope["sett"]


def _settings(sett):
    """Retain the original 130 settings/constants shared with maintained model."""
    result = {
        name: value for name, value in sett.__dict__.items()
        if name not in {"saltOcn_ref", "veros_fill"}
    }
    if len(result) != 130:
        raise RuntimeError("ERROR original shared settings inventory changed")
    return result


def _snapshot(arrays, case, step, state):
    """Freeze every maintained State name from the original mutable PyTree."""
    for name in STATE_FIELDS:
        arrays[f"{case}.state.{step}.{name}"] = np.asarray(getattr(state, name)).copy()


def _growth_transition(example_root):
    """Observe all original outputs while preserving notebook's ignored salt."""
    from veris.growth import Growth

    code = _notebook_function(example_root, "run_growth.ipynb", "run_growth")
    tree = ast.parse(code)
    function = next(node for node in tree.body if isinstance(node, ast.FunctionDef))
    assignment = next(node for node in function.body if isinstance(node, ast.Assign))
    output = assignment.value
    if not isinstance(output, ast.Call) or output.func.id != "Growth":
        raise RuntimeError("ERROR original growth update changed")
    index = function.body.index(assignment)
    function.body.insert(index, ast.Assign(
        targets=[ast.Name(id="captured_outputs", ctx=ast.Store())], value=output
    ))
    assignment.value = ast.Name(id="captured_outputs", ctx=ast.Load())
    returned = next(node for node in function.body if isinstance(node, ast.Return))
    returned.value = ast.Tuple(elts=[
        returned.value, ast.Name(id="captured_outputs", ctx=ast.Load())
    ], ctx=ast.Load())
    ast.fix_missing_locations(tree)
    scope = {"jax": jax, "partial": partial, "Growth": Growth}
    exec(compile(tree, str(example_root / "run_growth.ipynb"), "exec"), scope)
    return scope["run_growth"]


def _dynamics_transition(example_root):
    """Execute the unchanged full original dynamics notebook stage sequence."""
    imports = {
        "veris.area_mass": ("SeaIceMass", "AreaWS"),
        "veris.dynsolver": ("WindForcingXY", "IceVelocities"),
        "veris.dynamics_routines": ("SeaIceStrength",),
        "veris.ocean_stress": ("OceanStressUV",),
        "veris.advection": ("Advection",),
        "veris.clean_up": ("clean_up_advection", "ridging"),
        "veris.fill_overlap": ("fill_overlap",),
    }
    scope = {"jax": jax, "partial": partial}
    for module_name, names in imports.items():
        module = importlib.import_module(module_name)
        scope.update({name: getattr(module, name) for name in names})
    code = _notebook_function(example_root, "run_dyn.ipynb", "dyn_model")
    exec(compile(code, str(example_root / "run_dyn.ipynb"), "exec"), scope)
    return scope["dyn_model"]


def capture(model_root, example_root, output_root):
    """Capture independently hashed original notebook startup and trajectories."""
    model_root, example_root, output_root = (
        path.resolve() for path in (model_root, example_root, output_root)
    )
    _verify(model_root, MODEL_HASHES)
    _verify(example_root, EXAMPLE_HASHES)
    archive = output_root / "standalone_notebooks_v1.npz"
    manifest_path = output_root / "standalone_notebooks_v1.json"
    if archive.exists() or manifest_path.exists():
        raise RuntimeError("ERROR refusing to overwrite immutable reference outputs")
    sys.path[:0] = [str(model_root), str(example_root)]
    jax.config.update("jax_enable_x64", True)
    import veris
    from veris.settings import settings

    if not Path(veris.__file__).resolve().is_relative_to(model_root):
        raise RuntimeError("ERROR reference import escaped archived model source")
    if jax.default_backend() != "cpu":
        raise RuntimeError("ERROR original notebook reference capture requires CPU")
    arrays = {}
    cases = []
    state, sett = _initial_state(example_root, "initialize_growth.py")
    _snapshot(arrays, "growth", 0, state)
    history = {name: [] for name in GROWTH_OUTPUT_FIELDS}
    transition = _growth_transition(example_root)
    for _ in range(150):
        state, outputs = transition(state, sett)
        jax.block_until_ready(outputs)
        for name, value in zip(GROWTH_OUTPUT_FIELDS, outputs, strict=True):
            history[name].append(np.asarray(value)[0, 0])
    for name, values in history.items():
        arrays[f"growth.outputs.{name}"] = np.asarray(values)
    cases.append({
        "name": "growth", "grid": [1, 1], "storage_shape": [1, 1],
        "steps": 150, "snapshot_steps": [0], "output_steps": [1, 150],
        "timestep_seconds": sett.deltatTherm,

        "forcing_policy": "initialized atmospheric/ocean fields; recursive Qnet and Qsw",
        "state_updates": [name for name in GROWTH_OUTPUT_FIELDS if name != "forc_salt_surface"],
        "plot_policy": "notebook plots pre-step ice thickness; archive outputs are post-step",
        "adaptations": ["enable JAX float64 globally", "observe ignored output index 5 without applying it"],
        "execution_note": "original use_sharding=True is unused by the growth-only kernel",
    })
    print("PASS captured original growth: 150 days, eleven outputs", flush=True)
    settings["use_sharding"] = False
    state, sett = _initial_state(example_root, "initialize_dyn.py", dynamics=True)
    _snapshot(arrays, "dynamics", 0, state)
    transition = _dynamics_transition(example_root)
    for step in range(1, 101):
        state = transition(state, sett)
        jax.block_until_ready(state)
        if step in (1, 3, 100):
            _snapshot(arrays, "dynamics", step, state)
            for name in ("OceanStressU", "OceanStressV"):

                    getattr(state, name)

        "timestep_seconds": sett.deltatDyn,

        "forcing_policy": "fixed initialized wind snapshot 15 and ocean current",
        "adaptations": ["grid_len: 1024 -> 16", "use_circular_overlap: False -> True (notebook instruction)", "use_sharding: True -> False (notebook instruction)", "enable JAX float64 globally"],
        "halo_policy": "original notebook refreshes only hIceMean, hSnowMean and Area after each full step",
    })
    if not all(np.isfinite(value).all() for value in arrays.values()):
        raise RuntimeError("ERROR original notebook capture has nonfinite outputs")
    imported_sources = {
        name: str(Path(module.__file__).resolve().relative_to(model_root))
        for name, module in sys.modules.items()
        if name.startswith("veris") and getattr(module, "__file__", None)
    }
    output_root.mkdir(parents=True, exist_ok=True)
    np.savez_compressed(archive, **arrays)

            "veris": {"branch": "jax-only", "commit": MODEL_COMMIT, "source_hashes": MODEL_HASHES},

Hardcoded hashes validate both archived inputs before import.",

    print(f"PASS captured two original notebooks, {len(arrays)} finite float64 arrays", flush=True)


if __name__ == "__main__":
